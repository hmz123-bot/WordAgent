  }
}
DELETE /api/v1/documents/{id} 删除文档 
Response 200: { "code": 200, "data": null, "message": "文档已删除" }

📋 文档节点

GET /api/v1/documents/{id}/nodes 获取文档节点树 
Response 200:
{
  "code": 200,
  "data": [
    {
      "nodeId": "node-001",
      "type": "paragraph",
      "text": "第一章 项目背景",
      "level": 1,
      "structuralPath": "/1",
      "children": [
        {
          "nodeId": "node-002",
          "type": "paragraph",
          "text": "1.1 市场分析"
        }
      ]
    }
  ]
}
GET /api/v1/nodes/{id}/children 获取节点子节点 
Response 200: { "code": 200, "data": [ { "nodeId": "node-003", "type": "paragraph", "text": "1.1.1 市场规模" } ] }
PUT /api/v1/nodes/{id}/text 更新节点文本 
Request: { "text": "更新后的文本内容" }
Response 200: { "code": 200, "data": { "nodeId": "node-001", "text": "更新后的文本内容" } }
PUT /api/v1/nodes/{id}/attributes 更新节点属性 
Request: { "attributes": { "fontSize": 14, "bold": true } }
Response 200: { "code": 200, "data": { "nodeId": "node-001", "attributes": { "fontSize": 14 } } }
DELETE /api/v1/nodes/{id} 删除节点 
Response 200: { "code": 200, "data": null, "message": "节点已删除" }
POST /api/v1/nodes/{id}/insertAfter 在节点后插入新节点 
Request: { "type": "paragraph", "text": "新段落内容" }
Response 200: { "code": 200, "data": { "nodeId": "node-new", "text": "新段落内容" } }

          "position": 0

Response 200: { "code": 200, "data": { "totalReplaced": 3 } }

❑ 变更集

GET /api/v1/documents/{id}/changesets 获取变更集列表 
Response 200:
{
  "code": 200,
  "data": [
    {
      "changeSetId": "cs-001",
      "documentId": "doc-001",
      "changes": [ ... ],
      "status": "pending",
      "createdAt": "2024-12-01T10:30:00Z"
    }
  ]
}
POST /api/v1/documents/{id}/changesets 创建变更集 
Request: { "changes": [ { "type": "text_update", "nodeId": "node-001", "newText": "..." } ] }
Response 200: { "code": 200, "data": { "changeSetId": "cs-001", "status": "pending" } }
POST /api/v1/documents/{id}/changesets/{csId}/accept 接受变更集 
Response 200: { "code": 200, "data": { "changeSetId": "cs-001", "status": "accepted" } }
POST /api/v1/documents/{id}/changesets/{csId}/reject 拒绝变更集 
Response 200: { "code": 200, "data": { "changeSetId": "cs-001", "status": "rejected" } }
